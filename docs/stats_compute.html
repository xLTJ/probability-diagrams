 36def phi(z):
 37    """
 38    Φ(z) = P(Z ≤ z) for the standard normal Z ~ N(0, 1).
 39
 40    **Parameters**
 41
 42    - `z` *(float)* — z-score; any real number.
 43
 44    **Prints**
 45
 46    `Phi(<z>) = <r>`
 47
 48    **Examples**
 49
 50    ```python
 51    phi(1.96)    # → Phi(1.96) = 0.975002
 52    phi(-1.96)   # → Phi(-1.96) = 0.024998
 53    ```
 54    """
 55    result = stats.norm.cdf(z)
 56    print(f"Phi({z}) = {result:.6f}")
 57    return result

 60def standardize(x, mu, sigma2):
 61    """
 62    Converts an observed value x to a z-score: z = (x − μ) / σ.
 63
 64    Used to express how many standard deviations x lies above or below the mean.
 65    Useful before calling `phi()` when working with a non-standard normal,
 66    and when applying the CLT to convert a sample mean to a z-score.
 67
 68    **Parameters**
 69
 70    - `x` *(float)* — observed value to convert.
 71    - `mu` *(float)* — mean μ of the distribution.
 72    - `sigma2` *(float)* — variance σ² — **NOT standard deviation**. Pass `36` to get σ = 6.
 73
 74    **Prints**
 75
 76    `z = (<x> - <mu>) / <sigma> = <r>`
 77
 78    **Examples**
 79
 80    ```python
 81    standardize(8, mu=10, sigma2=36)   # → z = (8 - 10) / 6.0 = -0.333333
 82    ```
 83    """
 84    sigma = np.sqrt(sigma2)
 85    z = (x - mu) / sigma
 86    print(f"z = ({x} - {mu}) / {sigma:.4f} = {z:.6f}")
 87    return z

 94def normal_prob(mu, sigma2, a=-np.inf, b=np.inf):
 95    """
 96    P(a < X < b) for X ~ N(μ, σ²).
 97
 98    Frequently used in statistics for CLT approximations: if X̄ is a sample mean,
 99    then X̄ ~ N(μ, σ²/n) approximately, and this function computes probabilities
100    directly without needing to standardize first.
101
102    Provide one or both bounds. Omitting `a` gives a left-tail probability;
103    omitting `b` gives a right-tail probability.
104
105    **Parameters**
106
107    - `mu` *(float)* — mean μ.
108    - `sigma2` *(float)* — variance σ² — **NOT standard deviation**. Pass `36` to get σ = 6.
109    - `a` *(float, optional)* — lower bound; default −∞. Omit for P(X < b).
110    - `b` *(float, optional)* — upper bound; default +∞. Omit for P(X > a).
111
112    > **Note:** For continuous distributions, < and ≤ give identical results.
113
114    **Prints**
115
116    `P(<a> < X < <b>) for X ~ N(<mu>, <sigma2>) = <r>`
117
118    **Examples**
119
120    ```python
121    normal_prob(10, 36, b=5)     # P(X < 5)
122    normal_prob(10, 36, a=5)     # P(X > 5)
123    normal_prob(10, 36, 4, 16)   # P(4 < X < 16)
124    ```
125    """
126    sigma = np.sqrt(sigma2)
127    dist = stats.norm(loc=mu, scale=sigma)
128    result = dist.cdf(b) - dist.cdf(a)
129
130    a_str = "-inf" if a == -np.inf else str(a)
131    b_str = "+inf" if b == np.inf  else str(b)
132    print(f"P({a_str} < X < {b_str}) for X ~ N({mu}, {sigma2}) = {result:.6f}")
133    return result

140def binom_prob(n, p, a=None, b=None):
141    """
142    P(a ≤ X ≤ b) for X ~ Bin(n, p).
143
144    Provide one or both bounds. Omitting `a` gives P(X ≤ b);
145    omitting `b` gives P(X ≥ a).
146
147    **Parameters**
148
149    - `n` *(int)* — number of trials.
150    - `p` *(float)* — probability of success on each trial; must be in [0, 1].
151    - `a` *(int, optional)* — lower bound (inclusive). Omit for P(X ≤ b).
152    - `b` *(int, optional)* — upper bound (inclusive). Omit for P(X ≥ a).
153
154    **Prints**
155
156    `P(<a> <= X <= <b>) for X ~ Bin(<n>, <p>) = <r>`
157
158    **Examples**
159
160    ```python
161    binom_prob(10, 0.5, b=3)       # P(X ≤ 3)
162    binom_prob(10, 0.5, a=4)       # P(X ≥ 4)
163    binom_prob(10, 0.5, a=2, b=6)  # P(2 ≤ X ≤ 6)
164    ```
165    """
166    dist = stats.binom(n, p)
167
168    if a is None and b is not None:
169        result = dist.cdf(b)
170        print(f"P(X <= {b}) for X ~ Bin({n}, {p}) = {result:.6f}")
171    elif b is None and a is not None:
172        result = 1 - dist.cdf(a - 1)
173        print(f"P(X >= {a}) for X ~ Bin({n}, {p}) = {result:.6f}")
174    elif a is not None and b is not None:
175        result = dist.cdf(b) - dist.cdf(a - 1)
176        print(f"P({a} <= X <= {b}) for X ~ Bin({n}, {p}) = {result:.6f}")
177    else:
178        raise ValueError("Provide at least one of a or b.")
179
180    return result

213def sample_stats(data):
214    """
215    Computes key sample statistics from a dataset.
216
217    Uses **Bessel's correction** for sample variance (divides by n − 1, not n),
218    which gives an unbiased estimator of the population variance.
219
220    **Parameters**
221
222    - `data` *(list or numpy array)* — the raw observations.
223
224    **Prints**
225
226    ```
227    n        = <number of observations>
228    mean     = <x̄>
229    variance = <S²>    (divided by n-1)
230    std      = <S>
231    ```
232
233    **Examples**
234
235    ```python
236    sample_stats([2.1, 2.4, 1.9, 2.0, 2.3])
237    data = load_csv("temperature.csv", col=1, header=False)
238    sample_stats(data)
239    ```
240    """
241    data = np.asarray(data)
242    n    = len(data)
243    mean = np.mean(data)
244    var  = np.var(data, ddof=1)   # ddof=1 → divide by n-1 (Bessel's correction)
245    std  = np.sqrt(var)
246
247    print(f"n        = {n}")
248    print(f"mean     = {mean:.6f}")
249    print(f"variance = {var:.6f}  (divided by n-1)")
250    print(f"std      = {std:.6f}")
251
252    return n, mean, var, std

260def z_quantile(alpha, side='two'):
261    """
262    Standard normal quantile for CI construction or hypothesis testing.
263
264    - **Two-sided** (`side='two'`): returns Z_{α/2}, i.e. Φ⁻¹(1 − α/2).
265      Use for two-sided CIs and two-sided hypothesis tests.
266    - **One-sided** (`side='one'`): returns Z_α, i.e. Φ⁻¹(1 − α).
267      Use for one-sided hypothesis tests (upper- or lower-tail).
268
269    **Parameters**
270
271    - `alpha` *(float)* — significance level α; use 0.05 for a 95% CI or a
272      5% significance level test.
273    - `side` *(str, optional)* — `'two'` (default) or `'one'`.
274
275    **Prints**
276
277    `Z_{α/2} = <r>` or `Z_{α} = <r>` depending on `side`.
278
279    **Examples**
280
281    ```python
282    z_quantile(0.05)             # → Z_{0.025} = 1.959964  (two-sided, 95% CI)
283    z_quantile(0.05, side='one') # → Z_{0.05}  = 1.644854  (one-sided test)
284    z_quantile(0.01, side='one') # → Z_{0.01}  = 2.326348  (one-sided test)
285    ```
286    """
287    if side == 'one':
288        z = stats.norm.ppf(1 - alpha)
289        print(f"Z_{{α}} = {z:.6f}   (one-sided, α = {alpha})")
290    else:
291        z = stats.norm.ppf(1 - alpha / 2)
292        print(f"Z_{{α/2}} = {z:.6f}   (two-sided, α = {alpha})")
293    return z
294    return z

297def t_quantile(alpha, n=None, side='two', df=None):
298    """
299    t-distribution quantile for CI construction or hypothesis testing.
300
301    - **Two-sided** (`side='two'`): returns T_{α/2, df}, i.e. the value t
302      such that P(T > t) = α/2. Use for two-sided CIs and tests.
303    - **One-sided** (`side='one'`): returns T_{α, df}, i.e. the value t
304      such that P(T > t) = α. Use for one-sided hypothesis tests.
305
306    Degrees of freedom are set to n − 1 by default (CI and z-test context).
307    Pass `df` explicitly when a different value is needed — e.g. df = n − 2
308    for regression slope tests.
309
310    **Parameters**
311
312    - `alpha` *(float)* — significance level α; use 0.05 for a 95% CI.
313    - `n` *(int, optional)* — sample size; sets df = n − 1. Ignored if `df` is provided.
314    - `side` *(str, optional)* — `'two'` (default) or `'one'`.
315    - `df` *(int, optional)* — degrees of freedom. Overrides `n` when provided.
316      Use `df=n-2` for regression slope tests.
317
318    **Prints**
319
320    `T_{α/2, df} = <r>` or `T_{α, df} = <r>` depending on `side`.
321
322    **Examples**
323
324    ```python
325    t_quantile(0.05, n=9)              # → T_{0.025, 8} = 2.306004  (95% CI, n=9)
326    t_quantile(0.05, n=9, side='one')  # → T_{0.05,  8} = 1.859548  (one-sided)
327    t_quantile(0.05, df=10)            # → T_{0.025, 10}            (regression, n=12)
328    t_quantile(0.05, df=5, side='two') # → T_{0.025,  5}            (regression, n=7)
329    ```
330    """
331    if df is None:
332        if n is None:
333            raise ValueError("Provide either n or df.")
334        df = n - 1
335
336    if side == 'one':
337        t = stats.t.ppf(1 - alpha, df=df)
338        print(f"T_{{α, df}} = {t:.6f}   (df = {df}, one-sided, α = {alpha})")
339    else:
340        t = stats.t.ppf(1 - alpha / 2, df=df)
341        print(f"T_{{α/2, df}} = {t:.6f}   (df = {df}, two-sided, α = {alpha})")
342    return t

349def conf_interval(data, confidence=0.95, sigma2=None):
350    """
351    Confidence interval for the population mean μ.
352
353    Automatically selects the appropriate method based on what is known:
354
355    - **σ² known** (`sigma2` provided): uses Z_{α/2} with the true σ.
356      Exact when data are normal; approximate via CLT for large n.
357    - **σ² unknown** (`sigma2=None`): estimates variance from data using S²,
358      then uses T_{α/2, n−1}. Valid for any n when data are approximately normal;
359      for large n this converges to the Z result.
360
361    > **Note:** For large n with unknown σ², the t-CI and Z-CI (with S_n) give
362    > nearly identical results. The t-CI is always the safer choice.
363
364    **Parameters**
365
366    - `data` *(list or numpy array)* — the raw observations.
367    - `confidence` *(float, optional)* — confidence level 1 − α; default `0.95`.
368    - `sigma2` *(float or None, optional)* — known population variance σ².
369      Pass `None` (default) if σ² is unknown and should be estimated from data.
370
371    **Prints**
372
373    ```
374    Method   : Z  (known σ²)   or   t  (unknown σ², df = n−1)
375    n        : <sample size>
376    x̄        : <sample mean>
377    σ or S   : <std used>
378    quantile : <Z or T value>
379    CI       : (<lower>, <upper>)
380    half-width: <half-width>
381    ```
382
383    **Examples**
384
385    ```python
386    # Known variance
387    data = [5, 8.5, 12, 15, 7, 9, 7.5, 6.5, 10.5]
388    conf_interval(data, confidence=0.95, sigma2=4)
389
390    # Unknown variance — t method
391    conf_interval(data, confidence=0.95)
392
393    # Load from CSV first
394    data = load_csv("temperature.csv", col=1, header=False)
395    conf_interval(data, confidence=0.99)
396    ```
397    """
398    data  = np.asarray(data, dtype=float)
399    n     = len(data)
400    x_bar = np.mean(data)
401    alpha = 1 - confidence
402
403    if sigma2 is not None:
404        # Known variance — Z method
405        sigma    = np.sqrt(sigma2)
406        quantile = stats.norm.ppf(1 - alpha / 2)
407        method   = f"Z  (known σ² = {sigma2})"
408        std_used = sigma
409    else:
410        # Unknown variance — t method
411        s2       = np.var(data, ddof=1)
412        std_used = np.sqrt(s2)
413        quantile = stats.t.ppf(1 - alpha / 2, df=n - 1)
414        method   = f"t  (unknown σ², df = {n-1})"
415
416    half_width = quantile * std_used / np.sqrt(n)
417    lower      = x_bar - half_width
418    upper      = x_bar + half_width
419
420    label = "σ" if sigma2 is not None else "S"
421    print(f"Method    : {method}")
422    print(f"n         : {n}")
423    print(f"x̄         : {x_bar:.6f}")
424    print(f"{label}         : {std_used:.6f}")
425    print(f"quantile  : {quantile:.6f}")
426    print(f"CI        : ({lower:.6f}, {upper:.6f})")
427    print(f"half-width: {half_width:.6f}")
428
429    return lower, upper

436def regression_fit(x, y):
437    """
438    Least squares fit for the simple linear regression model Y = β₀ + β₁X + ε.
439
440    Computes and prints all intermediate quantities you need to verify your
441    hand calculations, then the final estimates and diagnostics.
442
443    **Printed output**
444
445    ```
446    n        = <number of pairs>
447    x̄        = <mean of x>
448    ȳ        = <mean of y>
449    s_xx     = Σ(xᵢ − x̄)²
450    s_xy     = Σ(xᵢ − x̄)(yᵢ − ȳ)
451    s_yy     = Σ(yᵢ − ȳ)²
452    β̂₁       = s_xy / s_xx
453    β̂₀       = ȳ − β̂₁ · x̄
454    σ̂²       = RSS / (n − 2)
455    R²       = s_xy² / (s_xx · s_yy)
456    residuals: [ε̂₁, ε̂₂, ...]
457    Σε̂ᵢ     = <should be ~0>
458    ```
459
460    **Parameters**
461
462    - `x` *(list or numpy array)* — predictor values.
463    - `y` *(list or numpy array)* — response values; must be the same length as `x`.
464
465    **Returns**
466
467    `(beta0, beta1, sigma2)` — the three fitted quantities most likely needed
468    for subsequent steps (slope test, prediction).
469
470    **Examples**
471
472    ```python
473    x = [0.41, 0.46, 0.44, 0.47, 0.42]
474    y = [1850, 2620, 2340, 2690, 2160]
475    beta0, beta1, sigma2 = regression_fit(x, y)
476    ```
477    """
478    x = np.asarray(x, dtype=float)
479    y = np.asarray(y, dtype=float)
480    n = len(x)
481
482    x_bar = np.mean(x)
483    y_bar = np.mean(y)
484    s_xx  = np.sum((x - x_bar) ** 2)
485    s_xy  = np.sum((x - x_bar) * (y - y_bar))
486    s_yy  = np.sum((y - y_bar) ** 2)
487
488    beta1 = s_xy / s_xx
489    beta0 = y_bar - beta1 * x_bar
490
491    residuals = y - (beta0 + beta1 * x)
492    rss       = np.sum(residuals ** 2)
493    sigma2    = rss / (n - 2)
494    r2        = s_xy ** 2 / (s_xx * s_yy)
495
496    print(f"n         = {n}")
497    print(f"x̄         = {x_bar:.6f}")
498    print(f"ȳ         = {y_bar:.6f}")
499    print(f"s_xx      = {s_xx:.6f}")
500    print(f"s_xy      = {s_xy:.6f}")
501    print(f"s_yy      = {s_yy:.6f}")
502    print(f"β̂₁        = {beta1:.6f}")
503    print(f"β̂₀        = {beta0:.6f}")
504    print(f"σ̂²        = {sigma2:.6f}  (RSS / (n-2))")
505    print(f"R²        = {r2:.6f}")
506    print(f"residuals : {np.round(residuals, 4).tolist()}")
507    print(f"Σε̂ᵢ       = {np.sum(residuals):.2e}  (should be ~0)")
508
509    return beta0, beta1, sigma2

512def regression_slope_test(x, y, beta_H=0, alpha=0.05, side='two'):
513    """
514    Hypothesis test for the slope β₁ in simple linear regression.
515
516    Tests H₀: β₁ = β_H (default 0) against H₁ determined by `side`.
517    Prints every intermediate quantity from the standard procedure so you
518    can verify each step of your hand calculation:
519
520    ```
521    β̂₁        = <slope>
522    σ̂²        = RSS / (n-2)
523    S_X       = sqrt(s_xx / n)
524    se(β̂₁)    = σ̂ / (S_X · √n)
525    T         = |β̂₁ − β_H| / se(β̂₁)
526    t_{α,df}  = <critical value>   (df = n-2)
527    p-value   = <p>
528    Decision  : Reject H₀  /  Fail to reject H₀
529    ```
530
531    **Parameters**
532
533    - `x` *(list or numpy array)* — predictor values.
534    - `y` *(list or numpy array)* — response values.
535    - `beta_H` *(float, optional)* — hypothesised slope value; default `0`.
536    - `alpha` *(float, optional)* — significance level; default `0.05`.
537    - `side` *(str, optional)* — `'two'` (default), `'upper'`, or `'lower'`.
538
539    **Examples**
540
541    ```python
542    # Ex 2 / fuel consumption example — test whether slope is zero
543    x = [45, 50, 55, 60, 65, 70, 75]
544    y = [24.2, 25.0, 23.3, 22.0, 21.5, 20.6, 19.8]
545    regression_slope_test(x, y)                        # H₀: β₁ = 0, two-sided
546
547    # Test against a specific non-zero slope
548    regression_slope_test(x, y, beta_H=0.5, alpha=0.01, side='upper')
549    ```

551    x = np.asarray(x, dtype=float)
552    y = np.asarray(y, dtype=float)
553    n = len(x)
554    df = n - 2
555
556    x_bar = np.mean(x)
557    y_bar = np.mean(y)
558    s_xx  = np.sum((x - x_bar) ** 2)
559    s_xy  = np.sum((x - x_bar) * (y - y_bar))
560    beta1 = s_xy / s_xx
561

565

573        reject = abs(T) > t_crit
574    elif side == 'upper':
575        t_crit = stats.t.ppf(1 - alpha, df=df)
576        p      = 1 - stats.t.cdf(T, df=df)
577        reject = T > t_crit
578    elif side == 'lower':
579        t_crit = stats.t.ppf(alpha, df=df)
580        p      = stats.t.cdf(T, df=df)
581        reject = T < t_crit
582    else:
583        raise ValueError("side must be 'two', 'upper', or 'lower'.")

586
587    print(f"β̂₁        = {beta1:.6f}")
588    print(f"σ̂²        = {sigma2:.6f}  (RSS / (n-2))")
589    print(f"S_X       = {S_X:.6f}  (sqrt(s_xx / n))")
590    print(f"se(β̂₁)    = {se:.6f}  (σ̂ / (S_X · √n))")
591    print(f"T         = {T:.6f}  ((β̂₁ − {beta_H}) / se)")
592    print(f"t_crit    = {t_crit:.6f}  (df = {df}, α = {alpha}, {side}-sided)")
593    print(f"p-value   = {p:.6f}  (t, df = {df})")
594    print(f"Decision  : {decision}  (α = {alpha})")

601    Confidence interval for the mean response and prediction interval at a
602    new predictor value x*, for the simple linear regression model.
603
604    Both intervals are centred on ŷ* = β̂₀ + β̂₁ · x*. The only difference
605    is the extra leading 1 inside the square root of the prediction interval,
606    which accounts for the irreducible noise in a single new observation.
607

609
610    ```
611    x*         = <x_star>
612    ŷ*         = β̂₀ + β̂₁ · x*
613    t_{α/2}    = <t-critical value, df = n-2>
614    CI (mean)  : (<lower>, <upper>)   ← for E[Y | X = x*]
615    PI (new)   : (<lower>, <upper>)   ← for a single new Y* at x*
616    ```
617
618    **Parameters**
619
620    - `x` *(list or numpy array)* — predictor values used to fit the model.
621    - `y` *(list or numpy array)* — response values used to fit the model.
622    - `x_star` *(float)* — new predictor value at which to predict.
623    - `alpha` *(float, optional)* — significance level; default 0.05 (95% intervals).
624
625    **Examples**
626
627    ```python
628    x = [0.41, 0.46, 0.44, 0.47, 0.42, 0.39, 0.41, 0.44, 0.43, 0.44]
629    y = [1850, 2620, 2340, 2690, 2160, 1760, 2500, 2750, 2730, 3120]
630    regression_predict(x, y, x_star=0.43)           # 95% PI and CI
631    regression_predict(x, y, x_star=3400, alpha=0.10)  # 90% CI
632    ```
633    """
634    x = np.asarray(x, dtype=float)
635    y = np.asarray(y, dtype=float)
636    n = len(x)
637
638    x_bar = np.mean(x)
639    y_bar = np.mean(y)
640    s_xx  = np.sum((x - x_bar) ** 2)
641    s_xy  = np.sum((x - x_bar) * (y - y_bar))
642    beta1 = s_xy / s_xx
643    beta0 = y_bar - beta1 * x_bar
644
645    residuals = y - (beta0 + beta1 * x)
646    sigma2    = np.sum(residuals ** 2) / (n - 2)
647
648    y_star  = beta0 + beta1 * x_star
649    t_crit  = stats.t.ppf(1 - alpha / 2, df=n - 2)
650
651    se_mean = np.sqrt(sigma2 * (1/n + (x_star - x_bar) ** 2 / s_xx))
652    se_pred = np.sqrt(sigma2 * (1 + 1/n + (x_star - x_bar) ** 2 / s_xx))
653
654    ci = (y_star - t_crit * se_mean, y_star + t_crit * se_mean)
655    pi = (y_star - t_crit * se_pred, y_star + t_crit * se_pred)
656
657    print(f"x*         = {x_star}")
658    print(f"ŷ*         = {y_star:.6f}")
659    print(f"t_{{α/2}}    = {t_crit:.6f}   (df = {n-2}, α = {alpha})")
660    print(f"CI (mean)  : ({ci[0]:.6f}, {ci[1]:.6f})")
661    print(f"PI (new)   : ({pi[0]:.6f}, {pi[1]:.6f})")
662
663    return ci, pi
